   from datetime import datetime, timezone
   try:
       called_at_dt = datetime.fromisoformat(row.called_at)

   elapsed = None
   if called_at_dt is not None:
       now = datetime.now(timezone.utc)
       if called_at_dt.tzinfo is None:

     function llmProgressCard() {
       return {

             case 'vibe_assign_pass1':         return '(Pass 1 — assigning tracks)';
             case 'vibe_assign_pass2':         return '(Pass 2 — boundary review)';

           try {

         },
       };
     }
