    """Phase 7 (Plan 03) — Suggestions queue endpoints.

    GET  /list                   — HTMX-friendly list re-render endpoint.
    """
    from __future__ import annotations

    def get_templates():
        from app.main import templates
        return templates


    @router.post("/{rating_key}/dismiss", response_class=HTMLResponse)
    async def dismiss_track(rating_key: str, request: Request):
        """Calls handle_dismiss_track (Plan 02) which writes NegativeSignal
        rows AND drains the mirror. Returns empty body — HTMX outerHTML swap
        from the dismiss button targets the row, so empty body == row removed.
        """
        from app.services.suggestions_service import handle_dismiss_track
        try:
            await handle_dismiss_track(rating_key)
        except Exception:
            logger.exception("dismiss_track failed for rating_key=%s", rating_key)
        return HTMLResponse(content="", status_code=200)


    @router.get("/list", response_class=HTMLResponse)
    async def suggestions_list(
        request: Request, session: Session = Depends(get_session)
    ):
        """Re-render the queue list. Used by /suggestions's HTMX poll on
        bootstrap-complete (signals that the empty-state loader should be
        replaced).
        """
        from app.models.suggestions import SuggestionsMirror
        from app.models.vibe import Vibe
        templates = get_templates()
        rows = session.exec(
            select(SuggestionsMirror)
            .order_by(SuggestionsMirror.position.asc())  # type: ignore[union-attr]
        ).all()
        enriched = []
        for r in rows:
            track = session.exec(
                select(Track).where(Track.id == r.track_id)
            ).first()
            vibe = None
            if r.vibe_id is not None:
                vibe = session.exec(
                    select(Vibe).where(Vibe.id == r.vibe_id)
                ).first()
            enriched.append({"row": r, "track": track, "vibe": vibe})

        """DEBUG-03 — Suggestions queue diagnostic page."""
