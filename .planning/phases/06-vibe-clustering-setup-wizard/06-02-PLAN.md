async def initial_cluster_proposal(forced_k: Optional[int] = None) -> VibeProposalSet
  1. Query rated tracks where all 4 audio features are non-null
  2. Z-score normalize the 4-D matrix (D-05; tempo cannot co-exist with 0-1 scales raw)
  3. If n_rated less than 30: return degraded_mode=True with single Your Taste proposal; SKIP LLM call
  4. Loop k in 3..k_max where k_max=min(7, n_rated // 15); pick silhouette argmax (Pitfall 3)
  5. forced_k overrides the loop; degraded_mode=True if best silhouette less than 0.25
  6. Build cacheable system prompt over 2048 tokens with taste profile + numbered track list (D-35)
  7. Call AnthropicClient.call_with_structured_output(... purpose=vibe_clustering_initial)
  8. Validate seed_track_indices in [0, n_rated); retry once on out-of-range
  9. Run materialize_clusters; return

async def refine_proposals(prior: VibeProposalSet, user_message: str, recluster_mode: bool = False) -> VibeProposalSet
  - Same cached system prompt as initial (within 1h TTL = cache hit)
  - User prompt = prior proposals JSON + user_message + Pydantic schema dump
  - purpose = vibe_clustering_recluster if recluster_mode else vibe_clustering_refine (D-34)
  - Validate + materialize like initial

def materialize_clusters(proposals: VibeProposalSet) -> VibeProposalSet
  Run constrained k-means per D-05:
  - keep / renamed_from: include tracks from source_vibe_ids
  - merged_from: union the source vibes' tracks; recompute single centroid
  - split_from: KMeans(k=2) within source vibe's tracks
  - new: KMeans seeded from seed_track_indices' centroid; pull additional tracks by distance
  - dropped: skip
  Returns same proposal list with centroid + spread + silhouette filled in per proposal.

async def slot_track(plex_rating_key) -> SlotInResult
  Per-track asyncio.Lock-guarded (D-16 / Pitfall 23). Inside the lock:
  1. Read latest Track row (NOT from event payload — Pitfall 23: always read latest from DB inside lock)
  2. If user_rating == 0 OR None: return without slotting (Phase 6 sees this as a no-op; unslot_track is the explicit cleared path)
  3. If energy / tempo / danceability / valence is None: SET pending_slot_in=TRUE; return SlotInResult(pending=True). D-17 retroactive path.
  4. Compute z-score-normalized 4-D distance from each Vibe.centroid (use Vibe.spread to scale per-dimension)
  5. closest_vibe = argmin; soft_2nd = vibes where distance ≤ closest_distance + 1*std-dev (Pitfall 24, cap at 2 — D-19 / VIBE-02)
  6. For each chosen vibe_id: SKIP if TrackVibe(track_id, vibe_id, assigned_by=manual) already exists (D-19 sticky)
  7. UPSERT TrackVibe(track_id, vibe_id, distance, assigned_at=now, assigned_by=auto-slot)
  8. For each affected vibe: get its ManagedPlaylist.plex_rating_key, then plex_playlist_service.update_playlist_items(playlist_key, all current member rating keys)  (additive — Pitfall 5)
  9. Return SlotInResult

async def unslot_track(plex_rating_key) -> None
  VIBE-10 + D-18. Rating cleared (10 to 0). Per-track lock applies.
  1. Find Track row + all current TrackVibe rows
  2. For each TrackVibe: look up vibe.ManagedPlaylist.plex_rating_key
  3. plex_playlist_service.remove_from_playlist for each affected playlist
  4. DELETE TrackVibe rows (all of them — including manual; rating cleared = full removal)

  2. Call slot_track(track.plex_rating_key)
  3. Clear pending_slot_in flag on success
