    # =====================================================================
    # Phase 8 Plan 04 — request-time helpers for /discover.
    # =====================================================================

    @dataclass
    class DiscoverSection:
        vibe: Vibe  # has .name, .color, .id
        candidates: list  # DiscoveryCandidate rows
        adds_in_flight: list  # DiscoveryAdd rows (vibe_slotted_at IS NULL)


    def _read_active_discover_data_sync() -> dict:
        """Returns the full data set /discover renders: {sections, lidarr_configured, vibes_exist, has_first_tick}."""
        from app.models.discovery import (
            DiscoveryCandidate, DiscoveryAdd, DiscoveryDismissed, WeeklyCronState,
        )
        from app.models.vibe import Vibe
        with Session(get_engine()) as session:
            vibes = session.exec(
                select(Vibe).where(Vibe.is_active == True).order_by(Vibe.id.asc())
            ).all()
            dismissed_mbids = {
                r for r in session.exec(select(DiscoveryDismissed.mb_id)).all()
            }
            completed_mbids = {
                r for r in session.exec(
                    select(DiscoveryAdd.mb_id).where(DiscoveryAdd.vibe_slotted_at.is_not(None))
                ).all()
            }
            in_flight_adds = session.exec(
                select(DiscoveryAdd).where(DiscoveryAdd.vibe_slotted_at.is_(None))
            ).all()
            in_flight_by_vibe: dict[int, list] = {}
            # in_flight_adds is keyed by mb_id but we want them grouped under
            # the seed_vibe_id of the original DiscoveryCandidate.
            cand_vibe_lookup = {
                c.mb_id: c.seed_vibe_id for c in session.exec(select(DiscoveryCandidate)).all()
            }
            for add in in_flight_adds:
                vid = cand_vibe_lookup.get(add.mb_id)
                if vid is not None:
                    in_flight_by_vibe.setdefault(vid, []).append(add)

            sections = []
            for vibe in vibes:
                cands = session.exec(
                    select(DiscoveryCandidate)
                    .where(DiscoveryCandidate.seed_vibe_id == vibe.id)
                    .where(DiscoveryCandidate.mb_id.not_in(dismissed_mbids or {""}))
                    .where(DiscoveryCandidate.mb_id.not_in(completed_mbids or {""}))
                    .order_by(DiscoveryCandidate.llm_rank.asc().nullslast())
                ).all()
                sections.append(DiscoverSection(
                    vibe=vibe,
                    candidates=cands,
                    adds_in_flight=in_flight_by_vibe.get(vibe.id, []),
                ))

            from app.services.settings_service import is_service_configured
            lidarr_configured = is_service_configured(session, "lidarr")

            weekly = session.get(WeeklyCronState, 1)
            has_first_tick = weekly is not None and weekly.last_tick_at is not None

        return {
            "sections": sections,
            "lidarr_configured": lidarr_configured,
            "vibes_exist": bool(vibes),
            "has_first_tick": has_first_tick,
        }


    async def read_active_discover_data() -> dict:
        return await asyncio.to_thread(_read_active_discover_data_sync)


    def _read_lidarr_settings_sync() -> Optional[dict]:
        """Returns {url, api_key, quality_profile_id, metadata_profile_id, root_folder_path} or None."""
        from app.services.settings_service import get_setting
        with Session(get_engine()) as session:
            cfg = get_setting(session, "lidarr")
            if cfg is None or not cfg.url or not cfg.api_key:
                return None
            extras = cfg.extra_config or {}
            try:
                return {
                    "url": cfg.url,
                    "api_key": cfg.api_key,  # decrypted at this layer per existing pattern
                    "quality_profile_id": int(extras.get("quality_profile_id") or 0),
                    "metadata_profile_id": int(extras.get("metadata_profile_id") or 0),
                    "root_folder_path": extras.get("root_folder_path") or "",
                }
            except (ValueError, TypeError):
                return None


    async def add_artist_to_lidarr(mb_id: str) -> dict:
        """D-D4 / DISC-05 — one-click Add handler.

        Reads quality + metadata profile + root from ServiceConfig.extras (Plan 01)
        and invokes lidarr_client.add_artist. On success, inserts a DiscoveryAdd
        row with added_at=now, lidarr_artist_id from response, lidarr_status="searching".
        Returns dict for the status-row partial render.
        """
        settings = await asyncio.to_thread(_read_lidarr_settings_sync)
        if settings is None:
            return {
                "success": False,
                "mb_id": mb_id,
                "error": "Lidarr not configured. Configure in Settings first.",
            }
        if settings["quality_profile_id"] == 0 or settings["metadata_profile_id"] == 0 or not settings["root_folder_path"]:
            return {
                "success": False,
                "mb_id": mb_id,
                "error": "Lidarr settings incomplete — re-test connection in Settings.",
            }

        from app.services import lidarr_client
        result = await lidarr_client.add_artist(
            mb_id=mb_id,
            url=settings["url"],
            api_key=settings["api_key"],
            quality_profile_id=settings["quality_profile_id"],
            metadata_profile_id=settings["metadata_profile_id"],
            root_dir=settings["root_folder_path"],
        )
        if not result.get("success"):
            return {"success": False, "mb_id": mb_id, "error": result.get("error", "Lidarr add failed.")}

        # Lookup the artist name from the most recent DiscoveryCandidate for breadcrumb display.
        artist_name = await asyncio.to_thread(_lookup_candidate_name_sync, mb_id) or mb_id
        await asyncio.to_thread(
            _insert_discovery_add_sync,
            mb_id, artist_name, result.get("lidarr_artist_id"),
        )
        return {
            "success": True,
            "mb_id": mb_id,
            "artist_name": artist_name,
            "lidarr_status": "searching",
        }


    def _insert_discovery_add_sync(mb_id: str, artist_name: str, lidarr_artist_id: Optional[int]) -> None:
        from app.models.discovery import DiscoveryAdd
        from datetime import datetime, timezone
        with Session(get_engine()) as session:
            session.add(DiscoveryAdd(
                mb_id=mb_id,
                artist_name=artist_name,
                added_at=datetime.now(timezone.utc).isoformat(),
                lidarr_artist_id=lidarr_artist_id,
                lidarr_status="searching",
            ))
            session.commit()


    def _lookup_candidate_name_sync(mb_id: str) -> Optional[str]:
        from app.models.discovery import DiscoveryCandidate
        with Session(get_engine()) as session:
            row = session.exec(
                select(DiscoveryCandidate.artist_name)
                .where(DiscoveryCandidate.mb_id == mb_id)
                .order_by(DiscoveryCandidate.id.desc())
            ).first()
            return row


    async def handle_dismiss_artist(mb_id: str) -> None:
        """D-D5 — artist-only exclude. Idempotent via UNIQUE(mb_id)."""
        from app.models.discovery import DiscoveryDismissed
        from sqlalchemy.exc import IntegrityError as _SAIntegrityError
        from datetime import datetime, timezone
        artist_name = await asyncio.to_thread(_lookup_candidate_name_sync, mb_id) or mb_id

        def _insert():
            with Session(get_engine()) as session:
                try:
                    session.add(DiscoveryDismissed(
                        mb_id=mb_id,
                        artist_name=artist_name,
                        dismissed_at=datetime.now(timezone.utc).isoformat(),
                    ))
                    session.commit()
                except _SAIntegrityError:
                    session.rollback()  # idempotent — already dismissed

        await asyncio.to_thread(_insert)


    # D-C3 — lazy 5-min cached Lidarr status poll. Module-level cache.
    _lidarr_status_cache: dict[str, tuple[str, datetime]] = {}
    _LIDARR_STATUS_CACHE_TTL = timedelta(minutes=5)


    async def get_lidarr_status_for_add(mb_id: str, lidarr_artist_id: Optional[int]) -> str:
        """D-C3 — lazy poll on /discover render with 5-min cache.

        Derives a human-readable status from Lidarr's history records.
        Returns one of: "pending" | "searching" | "downloading" | "imported, awaiting Composer sync" |
                        "analyzed, slotted into vibes" | "error".
        """
        now = datetime.now(timezone.utc)
        cached = _lidarr_status_cache.get(mb_id)
        if cached and (now - cached[1]) < _LIDARR_STATUS_CACHE_TTL:

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-08-19 | Tampering | Path-segment mb_id used in SQL inserts (DiscoveryDismissed.mb_id, DiscoveryAdd.mb_id) | mitigate | SQLModel parameterised inserts; never f-string into SQL. mb_id is a string with no special semantics for SQLite, so length and format-validation are belt-and-braces. Cap to first 64 chars in add/dismiss handlers (MBIDs are 36-char UUIDs; anything longer is a bug or attack). |
| T-08-20 | Tampering | path-segment mb_id used in Lidarr API call | mitigate | Lidarr's lookup_artist("mbid:<...>") parses the prefix; an attacker-controlled value that injects new query params would still go through pyarr's URL builder. pyarr uses the requests library which URL-encodes path/query.
