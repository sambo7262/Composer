    ```python
    # =====================================================================
    # Phase 8 Plan 05 — /debug/discovery data aggregation.
    # =====================================================================

    def _read_debug_discovery_data_sync() -> dict:
        from app.models.discovery import (
            DiscoveryCandidate, DiscoveryAdd, MusicBrainzCache,
        )
        from app.models.llm_usage import LLMUsage
        from app.models.event_log import EventLog
        from app.models.vibe import Vibe
        from sqlmodel import select, func

        with Session(get_engine()) as session:
            # 1) Last weekly candidate set with vibe joined for the header
            cand_rows = session.exec(
                select(DiscoveryCandidate).order_by(DiscoveryCandidate.created_at.desc()).limit(200)
            ).all()
            vibes_by_id = {v.id: v for v in session.exec(select(Vibe)).all()}
            candidates = [
                {
                    "mb_id": c.mb_id, "artist_name": c.artist_name,
                    "seed_vibe": vibes_by_id.get(c.seed_vibe_id),
                    "mb_listener_count": c.mb_listener_count,
                    "popularity_gate_pass": c.popularity_gate_pass,
                    "llm_rank": c.llm_rank, "factual_hook": c.factual_hook,
                    "llm_rationale": c.llm_rationale,
                    "created_at": c.created_at,
                }
                for c in cand_rows
            ]

            # 2) DiscoveryAdd lifecycle timeline (full history)
            adds = session.exec(
                select(DiscoveryAdd).order_by(DiscoveryAdd.added_at.desc()).limit(50)
            ).all()

            # 3) Recent MusicBrainz queries (via cache as proxy)
            mb_rows = session.exec(
                select(MusicBrainzCache).order_by(MusicBrainzCache.cached_at.desc()).limit(20)
            ).all()

            # 4) Recent Lidarr add_artist (DiscoveryAdd rows — same data as section 2,
            #    but framed as a Lidarr-request log)
            lidarr_adds = adds[:20]  # most recent 20

            # 5) Lidarr connection-test history (best-effort from EventLog).
            #    If no rows exist (project doesn't log test outcomes here), return [].
            try:
                test_history = session.exec(
                    select(EventLog)
                    .where(EventLog.source == "lidarr")
                    .order_by(EventLog.received_at.desc())
                    .limit(20)
                ).all()
            except Exception:
                test_history = []

            # Cost panel — artist-specific aggregate
            artist_cost_row = session.exec(
                select(func.coalesce(func.sum(LLMUsage.cost_estimate_usd), 0.0))
                .where(LLMUsage.purpose.like("discovery_artist_%"))
            ).first()
            artist_cost_usd = float(artist_cost_row or 0.0)

            artist_recent_calls = session.exec(
                select(LLMUsage)
                .where(LLMUsage.purpose.like("discovery_artist_%"))
                .order_by(LLMUsage.called_at.desc())
                .limit(20)
            ).all()

            # Counters from the most recent successful cron tick — best-effort.
            # If the cron didn't persist counts (Plan 02 returns CandidatePipelineCounts
            # but doesn't persist them yet), this section shows "Counters from last cron:
            # not persisted — see logs."
            # Inline TODO for future: add a DiscoveryCronCounts table if observability
            # demand grows.
            counts = None

        return {
            "candidates": candidates,
            "adds": adds,
            "mb_queries": mb_rows,
            "lidarr_adds": lidarr_adds,
            "lidarr_test_history": test_history,
            "artist_cost_usd": artist_cost_usd,
            "artist_recent_calls": artist_recent_calls,
            "counts": counts,
        }


    async def read_debug_discovery_data() -> dict:
        return await asyncio.to_thread(_read_debug_discovery_data_sync)
    ```
