    **B. app/services/poll_service.py** — NEW. Mirrors sync_service.py singleton pattern + uses bounded queries (D-08 + Pattern 3). Module-level _poll_status singleton, run_poll entry, _*_sync helpers wrapped via asyncio.to_thread:
    ```python
    from __future__ import annotations
    import asyncio
    import logging
    from dataclasses import dataclass, field
    from datetime import datetime, timezone
    from enum import Enum
    from typing import Optional, List

    from sqlmodel import Session, select
    from plexapi.server import PlexServer

    from app.database import get_engine
    from app.models.track import Track
    from app.models.events import RatingChangedEvent, TrackPlayedEvent, LibraryAddedEvent
    from app.services.event_bus import get_event_bus
    from app.services.settings_service import get_setting, get_decrypted_credential

    logger = logging.getLogger(__name__)


    class PollStateEnum(str, Enum):
        IDLE = "idle"
        RUNNING = "running"
        COMPLETED = "completed"
        FAILED = "failed"


    @dataclass
    class PollStatus:
        state: PollStateEnum = PollStateEnum.IDLE
        last_started: Optional[str] = None
        last_completed: Optional[str] = None
        last_changes_seen: int = 0
        error: Optional[str] = None


    _poll_status = PollStatus()


    def get_poll_status() -> PollStatus:
        return _poll_status


    def _get_existing_ratings_sync() -> dict:
        """Snapshot of {plex_rating_key: user_rating} from DB."""
        with Session(get_engine()) as session:
            rows = session.exec(select(Track.plex_rating_key, Track.user_rating)).all()
            return {row[0]: row[1] for row in rows}


    def _poll_recently_rated_sync(plex: PlexServer, library_id: int) -> List:
        """Bounded query — D-08, Pattern 3. NEVER library.all()."""
        section = plex.library.sectionByID(int(library_id))
        # Filter: tracks with userRating > 0; sorted by lastRatedAt; cap 200 (Pitfall 21).
        return section.searchTracks(
            filters={"track.userRating>>": 0},
            sort="lastRatedAt:desc",
            limit=200,
        )


    def _poll_recently_played_sync(plex: PlexServer, library_id: int) -> List:
        section = plex.library.sectionByID(int(library_id))
        return section.searchTracks(
            sort="lastViewedAt:desc",
            limit=200,
        )


    async def run_poll() -> None:
        """Plex polling pass. EVT-03. Bounded queries; emits typed events through shared bus."""
        global _poll_status
        if _poll_status.state == PollStateEnum.RUNNING:
            return  # concurrency guard, mirrors sync_service.py:152-159

        _poll_status = PollStatus(
            state=PollStateEnum.RUNNING,
            last_started=datetime.now(timezone.utc).isoformat(),
        )
        token = ""
        changes = 0
        try:
            with Session(get_engine()) as session:
                setting = get_setting(session, "plex")
                if not setting or not setting.is_configured:
                    raise ValueError("Plex not configured")
                token = get_decrypted_credential(session, "plex") or ""
                url = setting.url
                library_id = (setting.extra_config or {}).get("library_id")
                if not (url and token and library_id):
                    raise ValueError("Plex URL/token/library_id missing")

            # All PlexAPI calls via asyncio.to_thread (D-09)
            plex = await asyncio.to_thread(PlexServer, url, token, 30)
            existing = await asyncio.to_thread(_get_existing_ratings_sync)
            received_at = datetime.now(timezone.utc).isoformat()
            bus = get_event_bus()

            # 1. Rating diffs
            rated_tracks = await asyncio.to_thread(_poll_recently_rated_sync, plex, library_id)
            for t in rated_tracks:
                rk = str(t.ratingKey)
                new_rating = getattr(t, "userRating", None)
                if new_rating is None:
                    continue
                if existing.get(rk) != new_rating:
                    bus.put_nowait(
                        RatingChangedEvent(
                            plex_rating_key=rk,
                            new_rating=float(new_rating),
                            source="poll",
                            received_at=received_at,
                        )
                    )
                    changes += 1

            # 2. Play diffs
            played_tracks = await asyncio.to_thread(_poll_recently_played_sync, plex, library_id)
            for t in played_tracks:
                lv = getattr(t, "lastViewedAt", None)
                if lv is None:
                    continue
                rk = str(t.ratingKey)
                bus.put_nowait(
                    TrackPlayedEvent(
                        plex_rating_key=rk,

    from __future__ import annotations
    import asyncio
    import logging
    from dataclasses import dataclass
    from datetime import datetime, timezone
    from enum import Enum
    from typing import Optional

    from sqlmodel import Session, select

    from app.database import get_engine
    from app.models.track import Track
    from app.services.plex_client import get_library_tracks
    from app.services.settings_service import get_setting, get_decrypted_credential

    logger = logging.getLogger(__name__)


    class BackfillStateEnum(str, Enum):
        IDLE = "idle"
        RUNNING = "running"
        COMPLETED = "completed"
        FAILED = "failed"


    @dataclass
    class BackfillStatus:
        state: BackfillStateEnum = BackfillStateEnum.IDLE
        total_tracks: int = 0
        backfilled_tracks: int = 0
        last_started: Optional[str] = None
        last_completed: Optional[str] = None
        error: Optional[str] = None


    _backfill_status = BackfillStatus()


    def get_backfill_status() -> BackfillStatus:
        return _backfill_status


    def _check_needs_backfill_sync() -> bool:
        """Pitfall 7: tracks exist AND no track has user_rating populated."""
        with Session(get_engine()) as session:
            any_track = session.exec(select(Track.id).limit(1)).first()
            if not any_track:
                return False  # truly empty DB — sync hasn't run yet
            any_rated = session.exec(
                select(Track.id).where(Track.user_rating.is_not(None)).limit(1)  # type: ignore[attr-defined]
            ).first()
            return any_rated is None


    async def maybe_trigger_first_run_backfill() -> None:
        """Called from lifespan after start_scheduler. Idempotent."""
        global _backfill_status
        if _backfill_status.state == BackfillStateEnum.RUNNING:
            return
        needs = await asyncio.to_thread(_check_needs_backfill_sync)
        if needs:
            asyncio.create_task(run_backfill())


    def _upsert_rating_fields_sync(track_dicts: list[dict]) -> int:
        """Update user_rating, last_viewed_at, view_count for matching plex_rating_key.
        Returns count actually updated (existing tracks only — does NOT create new tracks here)."""
        updated = 0
        with Session(get_engine()) as session:
            for td in track_dicts:
                rk = td.get("plex_rating_key")
                if not rk:
                    continue
                stmt = select(Track).where(Track.plex_rating_key == rk)
                track = session.exec(stmt).first()
                if track is None:
                    continue  # track not in our DB; sync hasn't seen it yet
                track.user_rating = td.get("user_rating")
                track.last_viewed_at = td.get("last_viewed_at")
                track.view_count = td.get("view_count", 0)
                session.add(track)
                updated += 1
            session.commit()
        return updated


    async def run_backfill() -> None:
        """Page through Plex, populate user_rating/last_viewed_at/view_count on existing Track rows."""
        global _backfill_status
        if _backfill_status.state == BackfillStateEnum.RUNNING:
            return
        _backfill_status = BackfillStatus(
            state=BackfillStateEnum.RUNNING,
            last_started=datetime.now(timezone.utc).isoformat(),
        )
        token = ""
        try:
            with Session(get_engine()) as session:
                setting = get_setting(session, "plex")
                if not setting or not setting.is_configured:
                    raise ValueError("Plex not configured")
                token = get_decrypted_credential(session, "plex") or ""
                url = setting.url
                library_id = (setting.extra_config or {}).get("library_id")
                if not (url and token and library_id):
                    raise ValueError("Plex URL/token/library_id missing")

            batch_size = 200
            offset = 0
            total = None
            backfilled = 0
            while True:
                batch, batch_total = await get_library_tracks(
                    url, token, library_id,
                    container_start=offset, container_size=batch_size,
                )
                if total is None:
                    total = batch_total
                    _backfill_status.total_tracks = total
                count = await asyncio.to_thread(_upsert_rating_fields_sync, batch)
                backfilled += count
                _backfill_status.backfilled_tracks = backfilled
                # Yield control so HTMX banner status endpoint stays responsive
                await asyncio.sleep(0)
                if len(batch) < batch_size or (total is not None and offset + len(batch) >= total):
                    break
                offset += batch_size

            _backfill_status.state = BackfillStateEnum.COMPLETED
            _backfill_status.last_completed = datetime.now(timezone.utc).isoformat()
        except Exception as exc:
            _backfill_status.state = BackfillStateEnum.FAILED
            err = str(exc)
            if token:
                err = err.replace(token, "[REDACTED]")
            _backfill_status.error = err
            logger.exception("Backfill failed")
    ```
